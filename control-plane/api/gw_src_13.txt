            conn.execute(
                "UPDATE pending_actions SET status='rejected', approved_at=? WHERE id=? AND status='pending'",
                (time.time(), pid),
            )
        write_audit(
            "rejected",
            pending_action_id=pid,
            action="",
            target="",
            asset_id=payload.get("asset_id") or "",
            actor="admin",
            detail="marked benign / false positive",
            result_ok=0,
        )
    write_audit(
        "false_positive",
        action="mark_benign",
        target=payload.get("asset_id") or "",
        asset_id=payload.get("asset_id") or "",
        actor="admin",
        detail=rec.get("key"),
        result_ok=1,
    )
    return {"status": "ok", "record": rec}
