        cid = queue_agent_command(
            asset_id=asset_id,
            action="block_ip",  # agent uses restore; for unblock we send restore_network primarily
            target=str(ip),
            parameters={"undo": True, "hint": "operator_unblock"},
            reason=f"unblock {ip}",
        )
        # Prefer explicit restore only; block_ip with undo is a no-op unless agent handles it
        extra.append(cid)
    return {
        "status": "queued",
        "asset_id": asset_id,
        "command_id": cmd_id,
        "extra": extra,
        "note": "Agent will run restore_network (nft/iptables cyberguardian table remove). Requires root on host.",
    }


@app.post("/v1/feedback/benign")
async def mark_benign(payload: dict, x_api_key: str = Depends(require_admin)):
    """Mark detection fingerprint as benign — suppress restaging for N days."""
    title = payload.get("title") or ""
    if not title:
        raise HTTPException(400, "title required")
    rec = _fp.mark_benign(
        title=title,
        asset_id=payload.get("asset_id") or "",
        indicators=payload.get("indicators") or {},
        tags=payload.get("tags") or [],
        note=payload.get("note") or "",
        actor="admin",
        suppress_days=int(payload.get("suppress_days") or 14),
        detection_id=payload.get("detection_id") or "",
        pending_action_id=payload.get("pending_action_id") or "",
    )
    # If pending action id provided, reject it
    pid = payload.get("pending_action_id")
    if pid:
        with db() as conn:
            conn.execute(
                "UPDATE pending_actions SET status='rejected', approved_at=? WHERE id=? AND status='pending'",
                (time.time(), pid),
            )
        write_audit(
            "rejected",
            pending_action_id=pid,
            action="",
            target="",
            asset_id=payload.get("asset_id") or "",
            actor="admin",
            detail="marked benign / false positive",
            result_ok=0,
        )
    write_audit(
        "false_positive",
        action="mark_benign",
        target=payload.get("asset_id") or "",
        asset_id=payload.get("asset_id") or "",
        actor="admin",
        detail=rec.get("key"),
        result_ok=1,
    )
    return {"status": "ok", "record": rec}
